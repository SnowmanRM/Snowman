\chapter*{Glossary}

\begin{description}
	\item[HDD] Hard disk drive.
	\item[MD5] A cryptographic hash function producing a 128-bit hash value.
	\item[RAM] Random access memory.
	\item[RPM] Rounds per minute.
	\item[SSD] Solid state drive.
	\item[URL] Uniform resource locator.
\end{description}

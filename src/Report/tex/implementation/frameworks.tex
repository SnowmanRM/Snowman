\subsection{jQuery}
When it comes to using Javascript for client side scripting, jQuery\cite{jquery}
is the way to go. jQuery provides a framework for common javascript operations on the DOM
and abstracts away a lot of browser spesific code to give you a nice and clean
interface. It also makes AJAX class a lot easier, requireing only code such as
\begin{python}
\$.get('/web/rules/');
\end{python}
instead of 
\begin{python}
function loadXMLDoc() {
    var xmlhttp;

    if (window.XMLHttpRequest) {
        // code for IE7+, Firefox, Chrome, Opera, Safari
        xmlhttp = new XMLHttpRequest();
    } else {
        // code for IE6, IE5
        xmlhttp = new ActiveXObject("Microsoft.XMLHTTP");
    }

    xmlhttp.onreadystatechange = function() {
        if (xmlhttp.readyState == 4 && xmlhttp.status == 200) {
            document.getElementById("myDiv").innerHTML = xmlhttp.responseText;
        }
    }

    xmlhttp.open("GET", "/web/rules/", true);
    xmlhttp.send();
}

\end{python}
\subsection{Bootstrap}
